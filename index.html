// real entropy, in bits
const bits = len * Math.log2(pool);

           : bits < 50 ? 2 // padlock
           : bits < 70 ? 3 // deadbolt
           : 4;           // vault

// ALL four builds live on ONE
// timeline. The tier isn't a
// switch — it's a PLAYHEAD.
//
// Which is why a backspace
// doesn't reset the lock.
// It UN-BUILDS it.

  duration: .9,
